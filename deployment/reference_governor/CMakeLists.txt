cmake_minimum_required(VERSION 3.18)
project(franka_reference_governor VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
add_library(franka_reference_governor INTERFACE)
add_library(franka_governor::core ALIAS franka_reference_governor)
set_target_properties(franka_reference_governor PROPERTIES EXPORT_NAME core)
target_compile_features(franka_reference_governor INTERFACE cxx_std_17)
target_include_directories(franka_reference_governor INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(franka_reference_governor INTERFACE -ffp-contract=off -fno-fast-math)
elseif(MSVC)
  target_compile_options(franka_reference_governor INTERFACE /fp:strict)
endif()
option(GOVERNOR_PYTHON "Build Python binding" OFF)
if(GOVERNOR_PYTHON)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core src/bindings.cpp)
  target_link_libraries(_core PRIVATE franka_reference_governor)
  install(TARGETS _core LIBRARY DESTINATION franka_governor)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(governor_native_test tests/native_test.cpp)
  target_link_libraries(governor_native_test PRIVATE franka_reference_governor)
  add_test(NAME governor_native COMMAND governor_native_test)
endif()
install(TARGETS franka_reference_governor EXPORT franka_governorTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT franka_governorTargets NAMESPACE franka_governor:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/franka_governor)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/franka_governorConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/franka_governorConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/franka_governor)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/franka_governorConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/franka_governorConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/franka_governorConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/franka_governor)
if(GOVERNOR_PYTHON)
  install(DIRECTORY configs/ DESTINATION franka_governor/configs)
endif()
